import concurrent.futures
import hashlib
import json
from pathlib import Path
import urllib.request
from urllib.parse import quote

root = Path(__file__).parent
candidates = {
    'efficientnet': ('Darshan764__waste-classification-v2', ['model.weights.h5']),
    'siglip': ('prithivMLmods__Augmented-Waste-Classifier-SigLIP2', ['model.safetensors']),
    'mobilenet_detector': ('Ibrahim125__garbage-object-detection-fasterrcnn-mobilenetv3', ['model.pth']),
}

def fetch(item):
    name, (folder, filenames) = item
    source = root / 'research' / folder
    api = json.loads((source / 'api.json').read_text())
    dest = root / 'models' / name
    dest.mkdir(parents=True, exist_ok=True)
    for config in ['config.json', 'preprocessor_config.json', 'README.md']:
        if (source / config).exists():
            (dest / config).write_bytes((source / config).read_bytes())
    for filename in filenames:
        spec = next(s for s in api['siblings'] if s['rfilename'] == filename)
        path = dest / filename
        try:
            if not path.exists():
                url = f"https://huggingface.co/{api['id']}/resolve/{api['sha']}/{quote(filename)}"
                with urllib.request.urlopen(url, timeout=60) as response, path.with_suffix('.part').open('wb') as output:
                    while block := response.read(1024 * 1024):
                        output.write(block)
                path.with_suffix('.part').rename(path)
            digest = hashlib.sha256(path.read_bytes()).hexdigest()
            assert digest == spec['lfs']['sha256'] and path.stat().st_size == spec['size']
            print(name, 'verified', path.stat().st_size, digest, flush=True)
        except Exception as error:
            print(name, type(error).__name__, str(error), flush=True)

with concurrent.futures.ThreadPoolExecutor(max_workers=3) as pool:
    list(pool.map(fetch, candidates.items()))
